% Parameters of the motor drive inverter system 
R = 1; % Resistance (ohms) 
L = 0.1; % Inductance (henries) 
K = 0.01; % Motor constant 
J = 0.05; % Moment of inertia (kg.m^2) 
B = 0.1; % Damping coefficient 
Ke = 0.01; % Back EMF constant 
T_load = 0.1; % Load torque (N.m)  
% Define the transfer function model of the motor 
s = tf('s'); 
P_motor = K/((J*s+B)*(L*s+R)+K^2) 
% Define the state-space model of the motor 
A = [0 1 0; 0 -B/J K/J; 0 -Ke/L -R/L]; 
B = [0; 0; 1/L]; 
C = eye(3); 
D = zeros(3,1); 
motor_ss=ss(A,B,C,D) 
% Create a state-space model object 
sys = ss(A,B,C,D); 
% Define the simulation time and input signal 
t = 0:0.01:10; % Time vector 
u = ones(size(t)); % Input signal (e.g., voltage or torque command) 
% Simulate the system response 
[y,t,x] = lsim(sys, u, t); 
% Plot the results 
subplot(2,1,1); 
plot(t, y(:,1), 'b', t, y(:,2), 'r', t, y(:,3), 'g'); 
xlabel('Time (s)'); 
ylabel('Motor Variables'); 
legend('Position', 'Velocity', 'Current'); 
title('Motor Drive Inverter System Response');
